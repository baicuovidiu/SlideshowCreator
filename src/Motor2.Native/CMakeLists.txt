cmake_minimum_required(VERSION 3.24)
project(Motor2Native LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(Motor2Native SHARED src/d3d12_device.cpp)
target_compile_features(Motor2Native PRIVATE cxx_std_20)
target_include_directories(Motor2Native PUBLIC include)
set(NVENC_SDK_INCLUDE "" CACHE PATH "Path to NVIDIA Video Codec SDK Interface headers")
if(NVENC_SDK_INCLUDE)
  target_include_directories(Motor2Native PRIVATE "${NVENC_SDK_INCLUDE}")
  target_compile_definitions(Motor2Native PRIVATE MOTOR2_HAS_NVENC_SDK=1)
endif()
target_compile_definitions(Motor2Native PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Motor2Native PRIVATE d3d12 dxgi dxguid d3dcompiler)
